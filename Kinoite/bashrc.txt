# .bashrc

# Source global definitions
if [ -f /etc/bashrc ]; then
    . /etc/bashrc
fi

# User specific environment
if ! [[ "$PATH" =~ "$HOME/.local/bin:$HOME/bin:" ]]; then
    PATH="$HOME/.local/bin:$HOME/bin:$PATH"
fi
export PATH

# Uncomment the following line if you don't like systemctl's auto-paging feature:
# export SYSTEMD_PAGER=

# User specific aliases and functions
if [ -d ~/.bashrc.d ]; then
    for rc in ~/.bashrc.d/*; do
        if [ -f "$rc" ]; then
            . "$rc"
        fi
    done
fi
unset rc

alias cls='clear && fastfetch'
alias ls='LC_ALL=C ls -X --group-directories-first --color=auto'
alias dir='ls -l'
alias dira='ls -la'

alias ai_start='virsh -c qemu:///system start "Fedora Server"'
alias ai_stop='virsh -c qemu:///system shutdown "Fedora Server"'
alias ai_status='virsh -c qemu:///system domstate "Fedora Server"'
alias ai_console='virsh -c qemu:///system console "Fedora Server"'
alias ai='ssh -Y room10'

alias dev='distrobox enter'

BLUE='\[\e[38;5;81m\]'
ORANGE='\[\e[38;5;202m\]'
WHITE='\[\e[97m\]'
GREEN='\[\e[32m\]'
YELLOW='\[\e[33m\]'
RESET='\[\e[0m\]'

# Run only INSIDE a distrobox container
if [ -n "$CONTAINER_ID" ]; then
    export NVM_DIR="$HOME/.config/nvm"
    [ -s "$NVM_DIR/nvm.sh" ] && \. "$NVM_DIR/nvm.sh"  # This loads nvm
    [ -s "$NVM_DIR/bash_completion" ] && \. "$NVM_DIR/bash_completion"  # This loads nvm bash_completion
    . "$HOME/.cargo/env"

    PS1="📦 ${RESET}${BLUE}\u${RESET}${GREEN}@${RESET}${WHITE}\h ${ORANGE}\w ${GREEN}% ${WHITE}"
else
    export PATH="$PATH:/var/home/wmortl/Applications"
    
    alias ai_start='virsh -c qemu:///system start "Fedora Server"'
    alias ai_stop='virsh -c qemu:///system shutdown "Fedora Server"'
    alias ai_status='virsh -c qemu:///system domstate "Fedora Server"'
    alias ai_console='virsh -c qemu:///system console "Fedora Server"'
    alias ai='ssh -Y room10'

    alias dev='distrobox enter'

    PS1="🐧 ${RESET}${BLUE}\u${RESET}${GREEN}@${RESET}${WHITE}\h ${ORANGE}\w ${GREEN}% ${WHITE}"
fi
